pragma Singleton
import QtQuick
import Quickshell

// Path / file-type helpers for the Files app.
Singleton {
    function base(p) {
        const i = p.lastIndexOf("/");
        return i < 0 ? p : p.substring(i + 1);
    }
    function dirOf(p) {
        const i = p.lastIndexOf("/");
        return i <= 0 ? "/" : p.substring(0, i);
    }
    function ext(n) {
        const i = n.lastIndexOf(".");
        return i <= 0 ? "" : n.substring(i + 1).toLowerCase();
    }
    function isArchive(n) {
        return /\.(zip|7z|tar|tar\.gz|tgz|tar\.xz|txz|tar\.zst|tar\.bz2|tbz2)$/i.test(n);
    }
    function kind(e) {
        if (e.dir)
            return "folder";
        const x = ext(e.name);
        if (["png", "jpg", "jpeg", "webp", "gif", "bmp", "svg"].indexOf(x) >= 0)
            return "image";
        if (["mp3", "flac", "ogg", "wav", "m4a", "opus"].indexOf(x) >= 0)
            return "audio";
        if (["mp4", "mkv", "webm", "mov", "avi"].indexOf(x) >= 0)
            return "video";
        if (x === "pdf")
            return "pdf";
        if (isArchive(e.name))
            return "archive";
        return "text";
    }
    function icon(k) {
        return { folder: "folder", image: "image-x-generic", audio: "audio-x-generic", video: "video-x-generic", pdf: "application-pdf", archive: "package-x-generic", text: "text-x-generic" }[k];
    }
    function typeLabel(e) {
        const k = kind(e);
        if (k === "folder")
            return "Folder";
        const m = { image: "Image", audio: "Audio", video: "Video", pdf: "PDF document", archive: "Archive" };
        if (m[k])
            return m[k];
        const x = ext(e.name);
        return x === "" ? "File" : x.toUpperCase() + " file";
    }
    function human(b) {
        if (b < 1024)
            return b + " B";
        const u = ["KB", "MB", "GB", "TB"];
        let i = -1;
        do {
            b /= 1024;
            i++;
        } while (b >= 1024 && i < 3);
        return b.toFixed(b < 10 ? 1 : 0) + " " + u[i];
    }
    function date(t) {
        const d = new Date(t * 1000);
        return d.toDateString() === new Date().toDateString() ? Qt.formatTime(d, "HH:mm") : Qt.formatDate(d, "d MMM yyyy");
    }
}

# cpmv.sh MODE(cp|mv) DESTDIR SRC...  -> prints "src<TAB>dest" per item; never overwrites ("name (1).ext")
m=$1; d=$2; shift 2
for f in "$@"; do
  b=$(basename -- "$f"); n=$b; i=1
  [ "$m" = mv ] && [ "$(dirname -- "$f")" = "$d" ] && continue
  while [ -e "$d/$n" ]; do
    if [[ "$b" == *.* && "$b" != .* ]]; then n="${b%.*} ($i).${b##*.}"; else n="$b ($i)"; fi
    i=$((i+1))
  done
  if [ "$m" = cp ]; then cp -r -- "$f" "$d/$n"; else mv -- "$f" "$d/$n"; fi && printf '%s\t%s\n' "$f" "$d/$n"
done
EOF

    function menuAct(id) {
        const p = mctx.pane || activePane();
        if (!p)
            return;
        const es = p.selEntries();
        const one = es.length === 1 ? es[0] : null;
        const paths = p.sel.slice();
        if (id.indexOf("s:") === 0) {
            const k = id.substring(2);
            if (k === "desc")
                sortDesc = !sortDesc;
            else
                sortKey = k;
            return;
        }
        if (id.indexOf("cz:") === 0) {
            const fmt = id.substring(3);
            const suggested = paths.length === 1 ? Fx.base(paths[0]).replace(/\.[^.]*$/, "") || Fx.base(paths[0]) : "Archive";
            ask("Compress as ." + fmt, suggested, "Compress", name => compress(fmt, name, paths));
            return;
        }
        switch (id) {
        case "open": for (const e of es) p.open(e); break;
        case "newtab": if (one) newTab(one.path); break;
        case "inother": { const o = otherPane(); if (one && o) o.go(one.path); break; }
        case "openwith": if (one) { owPath = one.path; owQuery = ""; } break;
        case "cut": setClip(true); break;
        case "copy": setClip(false); break;
        case "paste": paste(); break;
        case "pasteinto": if (one && clipPaths.length) { copyMove(cutMode ? "mv" : "cp", clipPaths, one.path); if (cutMode) clipPaths = []; } break;
        case "rename": if (one) ask("Rename", one.name, "Rename", n => rename(one.path, n.trim())); break;
        case "copyother": { const o = otherPane(); if (o && paths.length) copyMove("cp", paths, o.path); break; }
        case "moveother": { const o = otherPane(); if (o && paths.length) copyMove("mv", paths, o.path); break; }
        case "compress": ctxMenu.popup(mctx.x, mctx.y, [{ id: "cz:zip", label: ".zip" }, { id: "cz:tgz", label: ".tar.gz" }, { id: "cz:txz", label: ".tar.xz" }, { id: "cz:7z", label: ".7z" }]); break;
        case "extract": if (one) extract(one.path); break;
        case "copypath": Quickshell.execDetached(["wl-copy", paths.join("\n")]); toast("Path copied"); break;
        case "bookmark": if (one && bookmarks.every(b => b.p !== one.path)) bookmarks = bookmarks.concat([{ n: one.name, p: one.path }]); break;
        case "rmbm": bookmarks = bookmarks.filter(b => b.p !== mctx.bm); break;
        case "props": if (one) showProps(one.path, one.name); break;
        case "folderprops": showProps(p.path, Fx.base(p.path) || "/"); break;
        case "trash": trashSel(paths); break;
        case "restore": restoreSel(paths); break;
        case "delperm": deleteForever(paths); break;
        case "emptytrash": emptyTrash(); break;
        case "newfolder": ask("New folder", "New folder", "Create", n => create(true, n)); break;
        case "newfile": ask("New text file", "New file.txt", "Create", n => create(false, n)); break;
        case "selall": p.selectAll(); break;
        case "invert": p.invert(); break;
        case "hidden": showHidden = !showHidden; break;
        case "single": singleClick = !singleClick; break;
        case "info": info = !info; break;
        case "split": toggleSplit(); break;
        case "undo": undo(); break;
        case "redo": redo(); break;
        case "refresh": refreshAll(); break;
        }
    }
