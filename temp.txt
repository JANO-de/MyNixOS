    **Safe execution.** No curl | sh. Argument arrays, never shell strings. No sudo in scripts.
    **Licences.** Keep upstream notices; meta.license truthful.
    **Secrets.** No tokens committed. GITHUB_TOKEN is read from the environment only, never stored.
    **Stop and ask** when: file/context missing; same error fails twice; destructive action not listed in the task; design choice not covered; scope creep. Items marked **ASK** always need the owner's answer first.

# Module loader: the only file that decides what gets imported from modules/.
# A folder with its own default.nix is one module; otherwise every *.nix file
# below is a module. Folders named pkg, src, dotfiles and pkgs hold packages
# and sources, not NixOS modules, so they are skipped. Order is alphabetical.
# Every module is off until a host enables it, e.g. modules.dev.java = true;
{ lib, ... }:
let
  skip = [
    "pkg"
    "src"
    "dotfiles"
    "pkgs"
  ];

  walk =
    dir:
    lib.concatLists (
      lib.mapAttrsToList (
        name: type:
        let
          path = dir + "/${name}";
        in
        if type == "directory" then
          if builtins.elem name skip then
            [ ]
          else if builtins.pathExists (path + "/default.nix") then
            [ path ]
          else
            walk path
        else if type == "regular" && lib.hasSuffix ".nix" name && name != "default.nix" then
          [ path ]
        else
          [ ]
      ) (builtins.readDir dir)
    );
in
{
  imports = walk ./.;
}

# Keep app-written configs inside the flake repo instead of loose files in ~/.config.
#
# ~/.config/<name> becomes a symlink to <repoPath>/hosts/<host>/dotfiles/<name> for every
# name in `names`. One folder per host, because some files are machine-specific.
# The links point at the live checkout (not the Nix store), so edits apply without a rebuild.
# `names` stays empty until the niri/mangowm/Noctalia step fills it.
{
  config,
  lib,
  inputs,
  ...
}:

let
  cfg = config.modules.desktop.dotfiles;
  hmDag = inputs.home-manager.lib.hm.dag;
  names = [ ];
in
{
  options.modules.desktop.dotfiles = {
    enable = lib.mkEnableOption "app configs stored in the flake repo via symlinks";
    repoPath = lib.mkOption {
      type = lib.types.str;
      default = "/home/jano/Proyectos/MyNixOS";
      description = "Absolute path of this repo's checkout on the machine (the links point here).";
    };
    host = lib.mkOption {
      type = lib.types.str;
      default = config.networking.hostName;
      defaultText = lib.literalExpression "config.networking.hostName";
      description = "Folder name under dotfiles/ used by this machine.";
    };
    dir = lib.mkOption {
      type = lib.types.str;
      default = "${cfg.repoPath}/hosts/${cfg.host}/dotfiles";
      description = "Directory the configs live in.";
    };
  };

  config = lib.mkIf cfg.enable {
    home-manager.users.jano =
      { config, ... }:
      {
        home.activation.dotfilesDirs = lib.mkIf (names != [ ]) (
          hmDag.entryBefore [ "linkGeneration" ] ''
            run mkdir -p ${lib.concatMapStringsSep " " (n: "\"${cfg.dir}/${n}\"") names}
          ''
        );
        xdg.configFile = lib.genAttrs names (n: {
          source = config.lib.file.mkOutOfStoreSymlink "${cfg.dir}/${n}";
        });
      };
  };
}
EOF

## Autostart

mangowm can automatically run commands or scripts upon startup. There are two modes for execution:

| Command | Behavior | Usage Case |
| :--- | :--- | :--- |
| exec-once | Runs **only once** when mangowm starts. | Status bars, Wallpapers, Notification daemons |
| exec | Runs **every time** the config is reloaded. | Scripts that need to refresh settings |

### Example Setup

# Start the status bar once
exec-once=waybar

# Set wallpaper
exec-once=swaybg -i ~/.config/mango/wallpaper/room.png

# Reload a custom script on config change
exec=bash ~/.config/mango/reload-settings.sh

# NixOS + Home-manager

The repo contains a flake that provides a NixOS module and a home-manager module for mango.
Use the NixOS module to install mango with other necessary components of a working Wayland environment.
Use the home-manager module to declare configuration and autostart for mango.

Here's an example of using the modules in a flake:

{
  inputs = {
    nixpkgs.url = "github:NixOS/nixpkgs/nixos-unstable";
    home-manager = {
      url = "github:nix-community/home-manager";
      inputs.nixpkgs.follows = "nixpkgs";
    };
    flake-parts.url = "github:hercules-ci/flake-parts";
    mango = {
      url = "github:mangowm/mango";
      inputs.nixpkgs.follows = "nixpkgs";
    };
  };
  outputs =
    inputs@{ self, flake-parts, ... }:
    flake-parts.lib.mkFlake { inherit inputs; } {
      debug = true;
      systems = [ "x86_64-linux" ];
      flake = {
        nixosConfigurations = {
          hostname = inputs.nixpkgs.lib.nixosSystem {
            system = "x86_64-linux";
            modules = [
              inputs.home-manager.nixosModules.home-manager

              # Add mango nixos module
              inputs.mango.nixosModules.mango
              {
                programs.mango.enable = true;
              }
              {
                home-manager = {
                  useGlobalPkgs = true;
                  useUserPackages = true;
                  backupFileExtension = "backup";
                  users."username".imports =
                    [
                      (
                        { ... }:
                        {
                          wayland.windowManager.mango = {
                            enable = true;
                            settings = ''
                              # see config.conf
                            '';
                            autostart_sh = ''
                              # see autostart.sh
                              # Note: here no need to add shebang
                            '';
                          };
                        }
                      )
                    ]
                    ++ [
                      # Add mango hm module
                      inputs.mango.hmModules.mango
                    ];
                };
              }
            ];
          };
        };
      };
    };
}
